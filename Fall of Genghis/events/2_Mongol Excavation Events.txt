	title = "The Excavation Mishap"
	desc = "81003D"
	picture = "ValleyoftheKings"

		has_country_modifier = mongological_excavations
		NOT = {
			OR = {
				owns = 1418
				tag = JAV
			}
		}
	}

	mean_time_to_happen = {
		months = 200
		modifier = {
			factor = 1.2
			is_greater_power = yes
		}
		modifier = {
			factor = 0.8
			NOT = {
				prestige = 50
			}
		}

			factor = 1.3
			nationalism_n_imperialism = 1
		}
	}

		name = "We are most sorry for this"

	}

	option = {

		relation = {
			who = JAV
			value = -20
		}
		clr_global_flag = mongol_excavation_rights
	}
}
