country_event = {
    id = 100951
    title = "Faction Status"
    desc = "100951D"
    picture = picture

    is_triggered_only = yes

    option = {
        name = "Should show loyalty of different factions"
        random_owned = {
            limit = {
                owner = {
                    check_variable = { which = egy_military_status value = 5 }
                }
            }
            owner = {
                egypt_military_loyal = {}
            }
        }
        random_owned = {
            limit = {
                owner = {
                    AND = {
                        check_variable = { which = egy_military_status value = -5 }
                        NOT = { check_variable = { which = egy_military_status value = 5 } }
                    }
                }
            }
            owner = {
                egypt_military_neutral = {}
            }
        }
        random_owned = {
            limit = {
                owner = {
                    NOT = { check_variable = { which = egy_military_status value = -4 } }
                }
            }
            owner = {
                egypt_military_disloyal = {}
            }
        }
    }
}
