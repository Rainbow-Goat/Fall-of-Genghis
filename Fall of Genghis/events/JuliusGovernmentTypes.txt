	id = 80220

	trigger = {
		not = { government = merchant_republic }
		or = {
			has_country_modifier = merchant_marine_funded
			has_country_modifier = free_market_embraced
			remove_country_modifier = merchant_marine_demands_submitted
			remove_country_modifier = merchant_marine_discontent
			remove_country_modifier = merchant_marine_dismantled
		}
	}

		months = 0.11

	picture = "News"

	option = {
		name = "I see"
		remove_country_modifier = merchant_marine_funded
		remove_country_modifier = free_market_embraced
		remove_country_modifier = merchant_marine_demands_submitted
		remove_country_modifier = merchant_marine_discontent
		remove_country_modifier = merchant_marine_dismantled
	}

country_event = {
	id = 80224
	title = "Merchant Marine demands friendlier government"
	desc = "Our support for the merchant marine has made them overconfident. Now they are starting to demand a more liberal government!"
	picture = "News"
	
	trigger = {
		has_country_modifier = merchant_marine_funded
		not = { ruling_party_ideology = liberal }
		not = { has_country_modifier = merchant_marine_discontent }
		not = { has_country_modifier = merchant_marine_demands_submitted }
	}
	
	mean_time_to_happen =  {
		months = 72
	}
	
	option = {
		name = "Give them what they demand!"
		ruling_party_ideology = liberal
		add_country_modifier = {
			name = merchant_marine_demands_submitted
			duration = 730
		}
		prestige = -5
	}
	
	option = {
		name = "Refuse their demands!"
		any_pop = {
			limit = { pop_majority_ideology = liberal }
			militancy = 2
		}
		add_country_modifier = {
			name = merchant_marine_discontent
			duration = 1825
		}
	}
	
	option = {
		name = "The time has come to revoke their special status!"
		remove_country_modifier = merchant_marine_funded
		add_country_modifier = {
			name = merchant_marine_dismantled
			duration = 3650
		}
		prestige = 5
	}
}
